# The cost benchmark: a program with a few thousand distinct instrumented
# functions and call sites spread over two shared libraries, built with -g,
# that reports the warm per-call cost of a traced call, the cold cost of the
# first lookups (the first call into each library, then every new address) and
# the resident memory after tracing. bench/run_bench.sh builds and runs it for
# every option combination against a checkout of your choice and prints one
# table, so that two checkouts (say master and a branch) can be compared on
# the same machine, the same day.
#
# Two ways to build it:
#   - as part of a top-level build: cmake -DBUILD_BENCH=ON ... ; then
#     `cmake --build build --target bench` builds and runs it;
#   - standalone, against ANY checkout (this is what run_bench.sh does):
#     cmake -S bench -B /tmp/bench -DCSLG_SOURCE_DIR=/path/to/checkout
#     [-DLOG_EXCEPTIONS=ON -DLOG_ELAPSED=ON -DCMAKE_BUILD_TYPE=RelWithDebInfo]

cmake_minimum_required(VERSION 3.14)
project(cslg_bench CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT TARGET callstacklogger)
    set(CSLG_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "The call-stack-logger checkout to measure")
    add_subdirectory(${CSLG_SOURCE_DIR} call-stack-logger)
endif()

set(CSLG_BENCH_FUNCTIONS 2000 CACHE STRING "Instrumented functions (and call sites) per library")

# Two generated shared libraries, each with CSLG_BENCH_FUNCTIONS tiny noinline
# functions and one driver function that calls every one of them from its own
# line — distinct callees and distinct call sites, the shape of real code.
# The libraries are instrumented like a plugin (the instrument flags plus -g,
# no link to the library): their hooks bind to the host's exported ones.
foreach(lib a b)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/bench_lib_${lib}.cpp")
    set(text "// Generated by bench/CMakeLists.txt — do not edit.\n")
    string(APPEND text "#define NOINLINE __attribute__((noinline))\n")
    math(EXPR last "${CSLG_BENCH_FUNCTIONS} - 1")
    foreach(i RANGE ${last})
        string(APPEND text "NOINLINE int bench_${lib}_${i}(int x) { __asm__ volatile(\"\"); return x + ${i}; }\n")
    endforeach()
    string(APPEND text "extern \"C\" int bench_${lib}_first(int x) { return bench_${lib}_0(x); }\n")
    string(APPEND text "extern \"C\" int bench_${lib}_run(int x) {\n    int sum = 0;\n")
    foreach(i RANGE ${last})
        string(APPEND text "    sum += bench_${lib}_${i}(x);\n")
    endforeach()
    string(APPEND text "    return sum;\n}\n")
    file(WRITE "${source}.new" "${text}")
    # Only touch the source when its content changed (keeps rebuilds incremental).
    execute_process(COMMAND ${CMAKE_COMMAND} -E copy_if_different "${source}.new" "${source}")
    file(REMOVE "${source}.new")

    add_library(bench_lib_${lib} SHARED "${source}")
    target_compile_options(bench_lib_${lib} PRIVATE
        $<TARGET_PROPERTY:callstacklogger_instrumented,INTERFACE_COMPILE_OPTIONS> -g)
endforeach()

add_executable(cslg_bench bench.cpp)
target_compile_definitions(cslg_bench PRIVATE CSLG_BENCH_FUNCTIONS=${CSLG_BENCH_FUNCTIONS})
target_link_libraries(cslg_bench PRIVATE callstacklogger::instrumented bench_lib_a bench_lib_b)

# `bench`: build and run once with the trace going to /dev/null, which is how
# the README's per-call numbers are defined (the write cost of a line to a
# real file depends on the file system, not on the tracer).
add_custom_target(bench
    COMMAND ${CMAKE_COMMAND} -E env CSLG_OUTPUT_FILE=/dev/null $<TARGET_FILE:cslg_bench>
    DEPENDS cslg_bench
    USES_TERMINAL
    COMMENT "Running the call-stack-logger cost benchmark (trace to /dev/null)")
